// Express Webhook Signature Verifier
import crypto from 'node:crypto';

app.post('/webhook', express.raw({ type: 'application/json' }), (req, res) => {
  const signature = req.headers['x-craftlanee-signature'];
  const secret = process.env.CRAFTLANEE_WEBHOOK_SECRET;

  const hmac = crypto.createHmac('sha256', secret);
  hmac.update(req.body);
  const expected = 'sha256=' + hmac.digest('hex');

  if (signature !== expected) {
    return res.status(401).send('Invalid signature');
  }

  const event = JSON.parse(req.body.toString());
  console.log('Verified Payment Settled Event:', event.order_id, event.utr);
  res.status(200).send('OK');
});
